        // =========================================================================
        // MODÈLE DU GALION PIRATE DÉTAILLÉ (SDF MULTI-ÉLÉMENTS)
        // =========================================================================
        float sdBox(vec3 p, vec3 b) {
            vec3 q = abs(p) - b;
            return length(max(q, 0.0)) + min(max(q.x, max(q.y, q.z)), 0.0);
        }

        float sdCappedCylinder(vec3 p, float h, float r) {
            vec2 d = abs(vec2(length(p.xz), p.y)) - vec2(r, h);
            return min(max(d.x, d.y), 0.0) + length(max(d, 0.0));
        }

        float sdSegment(vec3 p, vec3 a, vec3 b, float r) {
            vec3 pa = p - a, ba = b - a;
            float h = clamp(dot(pa, ba) / dot(ba, ba), 0.0, 1.0);
            return length(pa - ba * h) - r;
        }

        // Espace local du bateau :
        // z = axe proue(+) / poupe(-)
        // x = bâbord(-) / tribord(+)
        // y = vertical
        vec3 worldToShip(vec3 p) {
            vec3 q = p - u_ship_pos;
            q.xz *= rot2D(-u_ship_rot.x); // Yaw
            q.yz *= rot2D(-u_ship_rot.y); // Pitch
            q.xy *= rot2D(-u_ship_rot.z); // Roll
            return q;
        }

        // Bounding box du navire pour optimisation
        bool shipBoundingTest(vec3 ro, vec3 rd, out float tNear, out float tFar) {
            vec3 boxMin = u_ship_pos - vec3(6.5, 3.5, 6.5);
            vec3 boxMax = u_ship_pos + vec3(6.5, 8.5, 6.5);

            vec3 invD = 1.0 / rd;
            vec3 t0 = (boxMin - ro) * invD;
            vec3 t1 = (boxMax - ro) * invD;
            vec3 tSmall = min(t0, t1);
            vec3 tBig = max(t0, t1);

            tNear = max(max(tSmall.x, tSmall.y), tSmall.z);
            tFar = min(min(tBig.x, tBig.y), tBig.z);

            return (tFar >= max(tNear, 0.0));
        }

        // Matériaux du navire :
        // 1: Coque en bois de chêne vieilli / pont
        // 2: Ornements dorés de poupe et vitraux
        // 3: Voiles blanches bombées par le vent
        // 4: Mâts, vergues et beaupré (bois sombre)
        // 5: Pavillon pirate noir (Jolly Roger)
        // 6: Lanterne de poupe (lumière ambrée)
        // 7: Canons en bronze / fer forgé
        float map_ship_detail(vec3 q, out int matId) {
            matId = 1;
            float dMin = 1e5;

            // 1. COQUE DU GALION
            // Forme hydrodynamique effilée à l'avant, large au centre, château surélevé à l'arrière
            float zRel = q.z / 3.8;
            float bowTaper = smoothstep(0.0, 1.0, clamp(zRel, 0.0, 1.0));
            float sternTaper = smoothstep(0.0, 1.0, clamp(-zRel, 0.0, 1.0));
            
            float halfWidth = mix(1.25, 0.15, bowTaper) * mix(1.0, 0.85, sternTaper);
            float flare = 1.0 + (q.y + 0.5) * 0.28;
            halfWidth *= flare;

            // Hauteur du pont : s'élève à la proue (+0.4) et au château arrière (+0.85)
            float sheerLine = mix(0.65, 1.10, bowTaper * bowTaper) + mix(0.0, 0.85, sternTaper * sternTaper);
            
            // Coque extérieure
            float dHullX = abs(q.x) - halfWidth;
            float dHullY = max(q.y - sheerLine, -q.y - 0.95);
            float dHullZ = abs(q.z) - 3.8;
            float dHullOuter = max(dHullX, max(dHullY, dHullZ));

            // Arrondi de la carène
            float dKeelCurve = length(vec2(max(abs(q.x) - halfWidth * 0.45, 0.0), q.y + 0.7)) - 0.55;
            dHullOuter = max(dHullOuter, dKeelCurve);

            // Château arrière (cabine du capitaine)
            vec3 qCabin = q - vec3(0.0, 0.95, -2.5);
            float dCabin = sdBox(qCabin, vec3(0.95 * flare, 0.65, 1.15));
            dHullOuter = min(dHullOuter, dCabin);

            // Quille inférieure
            float dKeel = sdBox(q - vec3(0.0, -1.0, 0.0), vec3(0.08, 0.25, 3.7));
            dHullOuter = min(dHullOuter, dKeel);

            // Évidement du pont supérieur
            vec3 qDeck = q - vec3(0.0, sheerLine + 0.1, 0.0);
            float dDeckCut = sdBox(qDeck, vec3(halfWidth - 0.12, 0.35, 3.65));
            float dHull = max(dHullOuter, -dDeckCut);

            if (dHull < dMin) {
                dMin = dHull;
                matId = 1; // Bois de la coque
            }

            // 2. ORNEMENTS ET FENÊTRES DE POUPE
            vec3 qTransom = q - vec3(0.0, 1.05, -3.66);
            float dTransom = sdBox(qTransom, vec3(0.85, 0.45, 0.05));
            if (dTransom < dMin) {
                dMin = dTransom;
                matId = 2; // Ornements dorés
            }

            // 3. MÂTS, VERGUES ET BEAUPRÉ
            // Grand mât (central)
            float dMainMast = sdCappedCylinder(q - vec3(0.0, 3.1, 0.0), 3.4, 0.09);
            // Mât de misaine (avant)
            float dForeMast = sdCappedCylinder(q - vec3(0.0, 2.7, 1.9), 2.9, 0.08);
            // Mât d'artimon (arrière)
            float dMizzenMast = sdCappedCylinder(q - vec3(0.0, 2.8, -1.8), 2.5, 0.07);
            // Beaupré (pointant vers l'avant à 28°)
            vec3 qBowsprit = q - vec3(0.0, 0.8, 3.6);
            qBowsprit.yz *= rot2D(-0.48);
            float dBowsprit = sdCappedCylinder(qBowsprit - vec3(0.0, 1.3, 0.0), 1.4, 0.065);

            float dMasts = min(min(dMainMast, dForeMast), min(dMizzenMast, dBowsprit));

            // Vergues horizontales (traverses supportant les voiles)
            // Grand mât : 3 vergues
            float dYardM1 = sdCappedCylinder((q - vec3(0.0, 2.0, 0.0)).yxz, 0.045, 1.75);
            float dYardM2 = sdCappedCylinder((q - vec3(0.0, 3.8, 0.0)).yxz, 0.04, 1.35);
            float dYardM3 = sdCappedCylinder((q - vec3(0.0, 5.4, 0.0)).yxz, 0.035, 0.95);
            // Misaine : 2 vergues
            float dYardF1 = sdCappedCylinder((q - vec3(0.0, 1.8, 1.9)).yxz, 0.045, 1.45);
            float dYardF2 = sdCappedCylinder((q - vec3(0.0, 3.4, 1.9)).yxz, 0.038, 1.05);
            // Artimon : 2 vergues
            float dYardZ1 = sdCappedCylinder((q - vec3(0.0, 2.2, -1.8)).yxz, 0.04, 1.15);
            float dYardZ2 = sdCappedCylinder((q - vec3(0.0, 3.7, -1.8)).yxz, 0.035, 0.85);

            float dYards = min(min(min(dYardM1, dYardM2), min(dYardM3, dYardF1)), min(min(dYardF2, dYardZ1), dYardZ2));
            float dWoodRigging = min(dMasts, dYards);

            // Hunes (plateformes d'observation / nids de corbeau)
            float dHune1 = sdCappedCylinder(q - vec3(0.0, 3.65, 0.0), 0.10, 0.28) - 0.02;
            float dHune2 = sdCappedCylinder(q - vec3(0.0, 3.25, 1.9), 0.09, 0.24) - 0.02;
            dWoodRigging = min(dWoodRigging, min(dHune1, dHune2));

            if (dWoodRigging < dMin) {
                dMin = dWoodRigging;
                matId = 4; // Bois sombre mâts/vergues
            }

            // 4. VOILES CARRÉES BOMBÉES PAR LE VENT
            // Les voiles sont courbées par la brise tropicale
            float windBillow = 0.26;
            
            // Voile Grand Mât Inférieure
            vec3 qSM1 = q - vec3(0.0, 2.9, 0.0);
            float curveM1 = (1.0 - (qSM1.y / 0.85) * (qSM1.y / 0.85)) * (1.0 - (qSM1.x / 1.65) * (qSM1.x / 1.65)) * windBillow;
            float dSailM1 = max(sdBox(qSM1 - vec3(0.0, 0.0, curveM1), vec3(1.65, 0.82, 0.025)), -qSM1.y - 0.85);

            // Voile Grand Mât Supérieure
            vec3 qSM2 = q - vec3(0.0, 4.6, 0.0);
            float curveM2 = (1.0 - (qSM2.y / 0.72) * (qSM2.y / 0.72)) * (1.0 - (qSM2.x / 1.25) * (qSM2.x / 1.25)) * (windBillow * 0.9);
            float dSailM2 = max(sdBox(qSM2 - vec3(0.0, 0.0, curveM2), vec3(1.25, 0.70, 0.022)), -qSM2.y - 0.72);

            // Voile Misaine Inférieure
            vec3 qSF1 = q - vec3(0.0, 2.6, 1.9);
            float curveF1 = (1.0 - (qSF1.y / 0.75) * (qSF1.y / 0.75)) * (1.0 - (qSF1.x / 1.35) * (qSF1.x / 1.35)) * windBillow;
            float dSailF1 = max(sdBox(qSF1 - vec3(0.0, 0.0, curveF1), vec3(1.35, 0.72, 0.025)), -qSF1.y - 0.75);

            // Voile Misaine Supérieure
            vec3 qSF2 = q - vec3(0.0, 4.2, 1.9);
            float curveF2 = (1.0 - (qSF2.y / 0.65) * (qSF2.y / 0.65)) * (1.0 - (qSF2.x / 0.95) * (qSF2.x / 0.95)) * (windBillow * 0.85);
            float dSailF2 = max(sdBox(qSF2 - vec3(0.0, 0.0, curveF2), vec3(0.95, 0.62, 0.022)), -qSF2.y - 0.65);

            // Voile d'Artimon
            vec3 qSZ1 = q - vec3(0.0, 2.95, -1.8);
            float curveZ1 = (1.0 - (qSZ1.y / 0.68) * (qSZ1.y / 0.68)) * (1.0 - (qSZ1.x / 1.05) * (qSZ1.x / 1.05)) * windBillow;
            float dSailZ1 = max(sdBox(qSZ1 - vec3(0.0, 0.0, curveZ1), vec3(1.05, 0.65, 0.022)), -qSZ1.y - 0.68);

            // Foc triangulaire avant (entre beaupré et mât de misaine)
            vec3 qJib = q - vec3(0.0, 1.6, 3.2);
            float dJib = sdBox(qJib - vec3(0.0, 0.0, (1.0 - abs(qJib.y / 0.9)) * 0.18), vec3(0.025, 0.90, 0.95));

            float dSails = min(min(min(dSailM1, dSailM2), min(dSailF1, dSailF2)), min(dSailZ1, dJib));
            if (dSails < dMin) {
                dMin = dSails;
                matId = 3; // Toile des voiles
            }

            // 5. PAVILLON PIRATE (JOLLY ROGER) FLOTTANT AU VENT
            vec3 qFlag = q - vec3(0.0, 6.2, -0.45);
            float flagWave = sin(u_time * 7.0 + qFlag.z * 6.0) * 0.08 * smoothstep(0.0, -0.6, qFlag.z);
            float dFlag = sdBox(qFlag - vec3(flagWave, 0.0, 0.0), vec3(0.02, 0.22, 0.42));
            if (dFlag < dMin) {
                dMin = dFlag;
                matId = 5; // Pavillon noir pirate
            }

            // 6. LANTERNE DE POUPE DORÉE AVEC SOURCE LUMINEUSE
            vec3 qLantern = q - vec3(0.0, 1.65, -3.95);
            float dLanternFrame = sdCappedCylinder(qLantern, 0.22, 0.14) - 0.015;
            if (dLanternFrame < dMin) {
                dMin = dLanternFrame;
                matId = 6; // Lanterne brillante
            }

            // 7. CANONS SUR LES SABORDS (BABORD ET TRIBORD)
            vec3 qCannon = q;
            qCannon.z = mod(q.z + 1.2, 1.4) - 0.7;
            float cannonMask = step(abs(q.z), 2.2);
            if (cannonMask > 0.5) {
                float dGunLeft = sdCappedCylinder((q - vec3(-1.3, 0.35, q.z)).zxy, 0.35, 0.05);
                float dGunRight = sdCappedCylinder((q - vec3(1.3, 0.35, q.z)).zxy, 0.35, 0.05);
                float dCannons = min(dGunLeft, dGunRight);
                if (dCannons < dMin) {
                    dMin = dCannons;
                    matId = 7; // Bronze canons
                }
            }

            return dMin;
        }

        float map_ship_simple(vec3 q) {
            int dummyMat;
            return map_ship_detail(q, dummyMat);
        }

        // Normales du navire
        vec3 getShipNormal(vec3 q) {
            int dummy;
            float d = map_ship_detail(q, dummy);
            vec2 e = vec2(0.015, 0.0);
            return normalize(vec3(
                map_ship_detail(q + e.xyy, dummy) - d,
                map_ship_detail(q + e.yxy, dummy) - d,
                map_ship_detail(q + e.yyx, dummy) - d
            ));
        }

        // Raymarching du galion
        float raymarchShip(vec3 ro, vec3 rd, float maxDist, out int outMat, out vec3 outLocalP) {
            outMat = 0;
            outLocalP = vec3(0.0);

            float tNear, tFar;
            if (!shipBoundingTest(ro, rd, tNear, tFar)) return maxDist;

            float t = max(tNear - 0.1, 0.0);
            tFar = min(tFar + 0.1, maxDist);

            for (int i = 0; i < 70; i++) {
                vec3 p = ro + rd * t;
                vec3 q = worldToShip(p);
                int matId;
                float d = map_ship_detail(q, matId);

                if (d < 0.01) {
                    outMat = matId;
                    outLocalP = q;
                    return t;
                }
                t += d * 0.78;
                if (t > tFar) break;
            }
            return maxDist;
        }

        // Anti-aliased normal calculation with adaptive epsilon to suppress grid aliasing
        vec3 getSeaNormal(vec3 p, float eps) {
            vec3 n;
            n.y = map_sea_detailed(p);
            n.x = map_sea_detailed(vec3(p.x + eps, p.y, p.z)) - n.y;
            n.z = map_sea_detailed(vec3(p.x, p.y, p.z + eps)) - n.y;
            n.y = eps;
            return normalize(n);
        }

        vec3 getIslandNormal(vec3 p) {
            float d = map_island(p);
            vec2 e = vec2(0.02, 0.0);
            return normalize(vec3(
                map_island(p + e.xyy) - d,
                map_island(p + e.yxy) - d,
                map_island(p + e.yyx) - d
            ));
        }

        vec3 getSkyColor(vec3 rd, vec3 sunDir) {
            float sunDot = max(dot(rd, sunDir), 0.0);
            vec3 skyZenith  = vec3(0.02, 0.12, 0.38);
            vec3 skyHorizon = vec3(0.18, 0.42, 0.62);

            vec3 col = mix(skyHorizon, skyZenith, pow(max(rd.y, 0.0), 0.75));
            col += vec3(1.0, 0.80, 0.40) * pow(sunDot, 40.0) * 0.15;
            col += vec3(1.8, 1.7, 1.4) * pow(sunDot, 500.0);
            return col;
        }

        vec3 raymarchWater(vec3 ro, vec3 rd, out bool hit, out float outDist) {
            hit = false;
            outDist = 130.0;

            float t = 0.5;
            if (ro.y > 3.0 && rd.y < -0.01) {
                t = max(0.5, (3.0 - ro.y) / rd.y);
            }

            float t_prev = t;
            float dt = 0.25;

            for (int i = 0; i < 85; i++) {
                vec3 p = ro + rd * t;
                float h = map_sea(p);

                if (h < 0.0) {
                    float tA = t_prev;
                    float tB = t;
                    for (int j = 0; j < 6; j++) {
                        float tMid = 0.5 * (tA + tB);
                        vec3 pMid = ro + rd * tMid;
                        if (map_sea(pMid) < 0.0) {
                            tB = tMid;
                        } else {
                            tA = tMid;
                        }
                    }
                    hit = true;
                    outDist = 0.5 * (tA + tB);
                    return ro + rd * outDist;
                }

                t_prev = t;
                dt = 0.15 + t * 0.022;
                t += dt;
                if (t > 130.0) break;
            }

            outDist = 130.0;
            return ro + rd * 130.0;
        }

        float raymarchIsland(vec3 ro, vec3 rd, float maxDist) {
            float dO = 0.0;
            for (int i = 0; i < 75; i++) {
                vec3 p = ro + rd * dO;
                float dS = map_island(p);
                if (dS < 0.012 || dO > maxDist) break;
                dO += dS * 0.80;
            }
            return dO;
        }
